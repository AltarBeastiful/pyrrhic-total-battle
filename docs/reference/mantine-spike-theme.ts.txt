/**
 * Investigation 0007 — Pyrrhic's design tokens (`docs/design.md` §1) expressed as a Mantine theme.
 *
 * The friction this file exists to measure: our palette is *role*-shaped (each group has a `soft`
 * ground, a `strong` ink and an `edge`, different in light and dark), while Mantine wants a
 * ten-step lightness ramp per colour and picks one index out of it. The two models do not line up,
 * so every ramp here is *derived* from the two inks we actually own rather than authored.
 */

/** Each group's `strong` ink, light then dark (`docs/design.md` §1, "Group colours"). */

  specialists: ['#25568a', '#8fbcee'],

  monsters: ['#5b368a', '#c0a0ec'],

/* --- deriving a ten-step ramp from two inks ------------------------------------------------ */

function rgb(hex: string): [number, number, number] {
  const n = Number.parseInt(hex.slice(1), 16);
  return [(n >> 16) & 255, (n >> 8) & 255, n & 255];
}

function hex(c: [number, number, number]): string {
  return `#${c.map((v) => Math.round(Math.min(255, Math.max(0, v))).toString(16).padStart(2, '0')).join('')}`;
}

function mix(a: string, b: string, t: number): string {
  const [ar, ag, ab] = rgb(a);
  const [br, bg, bb] = rgb(b);
  return hex([ar + (br - ar) * t, ag + (bg - ag) * t, ab + (bb - ab) * t]);
}

/**
 * Mantine indexes 0 (lightest) → 9 (darkest). We own two inks — the one that reads on a light
 * surface and the one that reads on a dark one — so the ramp is built by mixing the dark-theme ink
 * up towards white for 0–4 and the light-theme ink down towards near-black for 6–9. Index 6 is the
 * light ink (Mantine's default filled shade) and index 3 the dark one, which is why `primaryShade`
 * below picks 6 in light and 3 in dark.
 */
function ramp(lightInk: string, darkInk: string): MantineColorsTuple {
  return [
    mix(darkInk, '#ffffff', 0.82),
    mix(darkInk, '#ffffff', 0.64),
    mix(darkInk, '#ffffff', 0.3),
    darkInk,
    mix(darkInk, lightInk, 0.35),
    mix(darkInk, lightInk, 0.7),
    lightInk,
    mix(lightInk, '#000000', 0.18),
    mix(lightInk, '#000000', 0.36),
    mix(lightInk, '#000000', 0.52),
  ] as unknown as MantineColorsTuple;
}

/** The accent is a metal, not a hue: dark Iron on light, struck Silver on dark. */
const metal = ramp(PYR.light.accent, PYR.dark.accent);

/** Mantine's own `dark` scale is what paints body, surfaces and borders in dark mode. */
const slate = [

  '#c3ccc8',
  PYR.dark.muted,
  '#6d7a76',
  '#4a5451',
  '#3a4341',
  PYR.dark.raised,
  PYR.dark.surface,
  PYR.dark.bg,
  '#101413',
] as unknown as MantineColorsTuple;

  primaryShade: { light: 6, dark: 3 },
  colors: {
    metal,
    dark: slate,
    guardsmen: ramp(...GROUP_INK.guardsmen),
    specialists: ramp(...GROUP_INK.specialists),
    engineers: ramp(...GROUP_INK.engineers),
    monsters: ramp(...GROUP_INK.monsters),
    mercenaries: ramp(...GROUP_INK.mercenaries),
  },

  // Fraunces sets numerals only (design.md §3): the roman tier and the one hero figure. Mantine has
  // no "numerals" slot, so headings carry it here and the `.numeral-face` class does the rest.
  headings: {
    fontFamily: "'Fraunces Variable', ui-serif, Georgia, 'Times New Roman', serif",
    fontWeight: '500',
  },
  defaultRadius: 'sm',

  autoContrast: true,
  luminanceThreshold: 0.4,
});
